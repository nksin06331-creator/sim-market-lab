```sh
python scripts/generate_coreweave_reports.py --materials /workspace/report-generation-materials
python /workspace/report-generation-materials/11_validate_report_bundle.py data/reports/coreweave-crwv.json --html stocks/coreweave-crwv/company.html stocks/coreweave-crwv/valuation.html stocks/coreweave-crwv/catalysts.html --pretty
```
